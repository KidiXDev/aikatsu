cmake_minimum_required(VERSION 3.21)
project(mpv_stats_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Quick Network OpenGL)
find_path(MPV_INCLUDE_DIR mpv/client.h HINTS "$ENV{MPV_DIR}/include" REQUIRED)
find_library(MPV_LIBRARY NAMES mpv HINTS "$ENV{MPV_DIR}" REQUIRED)
add_executable(mpv_stats_test mpv_stats_test.cpp ../cpp/mpvitem.cpp ../cpp/mpvitem.h)
target_include_directories(mpv_stats_test PRIVATE ../cpp ${MPV_INCLUDE_DIR})
target_link_libraries(mpv_stats_test PRIVATE Qt6::Quick Qt6::Network Qt6::OpenGL ${MPV_LIBRARY})
if(WIN32)
    target_link_libraries(mpv_stats_test PRIVATE dwmapi)
elseif(UNIX AND NOT APPLE)
    find_package(Qt6 REQUIRED COMPONENTS DBus)
    target_link_libraries(mpv_stats_test PRIVATE Qt6::DBus)
endif()
enable_testing()
add_test(NAME mpv_stats COMMAND mpv_stats_test)
set_tests_properties(mpv_stats PROPERTIES TIMEOUT 20 ENVIRONMENT "QSG_RENDER_LOOP=threaded")
